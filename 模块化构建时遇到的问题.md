`org.eclipse:org.eclipse.tm4e.core`依赖`org.jruby.joni:joni`，但是jdeps无法分析出来，所以`jdeps`的模块要额外手动添加到
`--add-modules`中。
使用``为`org.eclipse:org.eclipse.tm4e.core`写入`module-info.class`时，也要手动添加`requires org.jruby.joni;`。

```
<plugin>
    <groupId>org.moditect</groupId>
    <artifactId>moditect-maven-plugin</artifactId>
    <version>1.0.0.Final</version>
    <executions>
        <execution>
            <id>add-module-infos</id>
            <phase>generate-resources</phase>
            <goals>
                <goal>add-module-info</goal>
            </goals>
            <configuration>
                <outputDirectory>${project.build.directory}/modules</outputDirectory>
                <modules>
                    <module>
                        <artifact>
                            <groupId>org.eclipse</groupId>
                            <artifactId>org.eclipse.tm4e.core</artifactId>
                            <version>0.16.0-SNAPSHOT</version>
                        </artifact>
                        <moduleInfoSource>
                            module org.eclipse.tm4e.core {
                                requires org.jruby.joni;
                                requires com.google.gson;
                                requires org.snakeyaml.engine.v2;
                                requires java.xml;
                                requires jdk.xml.dom;
                                exports org.eclipse.tm4e.core;
                                exports org.eclipse.tm4e.core.grammar;
                                exports org.eclipse.tm4e.core.internal.grammar;
                                exports org.eclipse.tm4e.core.internal.grammar.dependencies;
                                exports org.eclipse.tm4e.core.internal.grammar.raw;
                                exports org.eclipse.tm4e.core.internal.grammar.tokenattrs;
                                exports org.eclipse.tm4e.core.internal.matcher;
                                exports org.eclipse.tm4e.core.internal.oniguruma;
                                exports org.eclipse.tm4e.core.internal.parser;
                                exports org.eclipse.tm4e.core.internal.registry;
                                exports org.eclipse.tm4e.core.internal.rule;
                                exports org.eclipse.tm4e.core.internal.theme;
                                exports org.eclipse.tm4e.core.internal.theme.css;
                                exports org.eclipse.tm4e.core.internal.theme.css.sac;
                                exports org.eclipse.tm4e.core.internal.theme.css.util;
                                exports org.eclipse.tm4e.core.internal.theme.raw;
                                exports org.eclipse.tm4e.core.internal.utils;
                                exports org.eclipse.tm4e.core.model;
                                exports org.eclipse.tm4e.core.registry;
                                exports org.eclipse.tm4e.core.theme;
                                exports org.eclipse.tm4e.core.theme.css;
                            }
                        </moduleInfoSource>
                    </module>
                </modules>
                <jdepsExtraArgs>
                    <arg>--ignore-missing-deps</arg>
                    <arg>--multi-release</arg>
                    <arg>9</arg>
                </jdepsExtraArgs>
            </configuration>
        </execution>
    </executions>
</plugin>
```
